   111→   111→async function createBooking(body: Record<string, unknown>) {
   112→   112→  const {
   113→   113→    roomId, phone, name, company, gst, address, checkIn, checkOut,
   114→   114→    guestCount, advance, isCorporate, notes,
   115→   115→  } = body
   116→   116→  if (!roomId || !phone || !name) {
   117→   117→    return NextResponse.json({ error: 'Room, phone and name are required' }, { status: 400 })
   118→   118→  }
   119→   119→  const room = await DB.room.findUnique({ where: { id: String(roomId) } })
   120→   120→  if (!room) return NextResponse.json({ error: 'Room not found' }, { status: 404 })
   121→   121→  if (room.status === 'OCCUPIED') {
   122→   122→    return NextResponse.json({ error: `Room ${room.number} is already occupied` }, { status: 400 })
   123→   123→  }
   124→   124→
   125→   125→  const guest = await DB.guest.upsert({
   126→   126→    where: { phone: String(phone) },
   127→   127→    update: {
   128→   128→      name: String(name),
   129→   129→      ...(company !== undefined && { company: String(company) }),
   130→   130→      ...(gst !== undefined && { gst: String(gst) }),
   131→   131→      ...(address !== undefined && { address: String(address) }),
   132→   132→    },
   133→   133→    create: {
   134→   134→      phone: String(phone),
   135→   135→      name: String(name),
   136→   136→      ...(company ? { company: String(company) } : {}),
   137→   137→      ...(gst ? { gst: String(gst) } : {}),
   138→   138→      ...(address ? { address: String(address) } : {}),
   139→   139→    },
   140→   140→  })
   141→   141→
   142→   142→  const checkInDate = checkIn ? new Date(String(checkIn)) : new Date()
   143→   143→  const checkOutDate = checkOut ? new Date(String(checkOut)) : null
   144→   144→  let days = 1
   145→   145→  if (checkOutDate) {
   146→   146→    const diff = Math.ceil((checkOutDate.getTime() - checkInDate.getTime()) / (1000 * 60 * 60 * 24))
   147→   147→    days = Math.max(1, diff)
   148→   148→  }
   149→   149→
   150→   150→  const booking = await DB.booking.create({
   151→   151→    data: {
   152→   152→      roomId: String(roomId),
   153→   153→      guestId: guest.id,
   154→   154→      checkIn: checkInDate,
   155→   155→      checkOut: checkOutDate,
   156→   156→      days,
   157→   157→      guestCount: parseInt(String(guestCount)) || 1,
   158→   158→      ratePerDay: room.rate,
   159→   159→      advance: parseFloat(String(advance)) || 0,
   160→   160→      isCorporate: !!isCorporate,
   161→   161→      notes: notes ? String(notes) : null,
   162→   162→    },
   163→   163→    include: { room: true, guest: true },
   164→   164→  })
   165→   165→
   166→   166→  await DB.room.update({ where: { id: String(roomId) }, data: { status: 'OCCUPIED' } })
   167→   167→
   168→   168→  const adv = parseFloat(String(advance)) || 0
   169→   169→  if (adv > 0) {
   170→   170→    await DB.ledgerEntry.create({
   171→   171→      data: {
   172→   172→        date: new Date(),
   173→   173→        type: 'INCOME',
   174→   174→        category: 'ADVANCE',
   175→   175→        description: `Advance from ${name} (Room ${room.number})`,
   176→   176→        amount: adv,
   177→   177→        method: 'CASH',
   178→   178→        source: 'AUTO',
   179→   179→        refId: booking.id,
   180→   180→      },
   181→   181→    })
   182→   182→  }
   183→   183→  return NextResponse.json(booking)
   184→   184→}

   186→   186→async function updateBooking(body: Record<string, unknown>) {
   187→   187→  const { id, action, checkOut } = body
   188→   188→  const booking = await DB.booking.findUnique({ where: { id: String(id) }, include: { room: true } })
   189→   189→  if (!booking) return NextResponse.json({ error: 'Booking not found' }, { status: 404 })
   190→   190→
   191→   191→  if (action === 'checkout') {
   192→   192→    const pendingFood = await DB.foodOrder.aggregate({
   193→   193→      where: { bookingId: String(id), status: 'PENDING' },
   194→   194→      _sum: { total: true },
   195→   195→    })
   196→   196→    if ((pendingFood._sum.total || 0) > 0) {
   197→   197→      return NextResponse.json(
   198→   198→        { error: 'Pending food orders exist. Add them to bill or mark paid first.' },
   199→   199→        { status: 400 }
   200→   200→      )
   201→   201→    }
   202→   202→    const updated = await DB.booking.update({
   203→   203→      where: { id: String(id) },
   204→   204→      data: { status: 'COMPLETED', actualCheckOut: new Date() },
   205→   205→    })
   206→   206→    await DB.room.update({ where: { id: booking.roomId }, data: { status: 'VACANT' } })
   207→   207→    return NextResponse.json(updated)
   208→   208→  }
   209→   209→
   210→   210→  if (action === 'extend') {
   211→   211→    const newCheckOut = checkOut ? new Date(String(checkOut)) : null
   212→   212→    if (!newCheckOut) return NextResponse.json({ error: 'New checkout date required' }, { status: 400 })
   213→   213→    const diff = Math.ceil(
   214→   214→      (newCheckOut.getTime() - new Date(booking.checkIn).getTime()) / (1000 * 60 * 60 * 24)
   215→   215→    )
   216→   216→    const days = Math.max(1, diff)
   217→   217→    const updated = await DB.booking.update({
   218→   218→      where: { id: String(id) },
   219→   219→      data: { checkOut: newCheckOut, days },
   220→   220→    })
   221→   221→    return NextResponse.json(updated)
   222→   222→  }
   223→   223→
   224→   224→  if (action === 'cancel') {
   225→   225→    const updated = await DB.booking.update({
   226→   226→      where: { id: String(id) },
   227→   227→      data: { status: 'CANCELLED', actualCheckOut: new Date() },
   228→   228→    })
   229→   229→    await DB.room.update({ where: { id: booking.roomId }, data: { status: 'VACANT' } })
   230→   230→    return NextResponse.json(updated)
   231→   231→  }
   232→   232→
   233→   233→  return NextResponse.json({ error: 'Unknown action' }, { status: 400 })
   234→   234→}

   255→   255→async function createBill(body: Record<string, unknown>) {
   256→   256→  const {
   257→   257→    bookingId, days, billedRoomTotal, gstPercent, extraCharges, discount,
   258→   258→    payCash, payUpi, payCard, includeFood, corporateName, gstNumber, notes, checkout,
   259→   259→  } = body
   260→   260→
   261→   261→  const booking = await DB.booking.findUnique({
   262→   262→    where: { id: String(bookingId) },
   263→   263→    include: { room: true, guest: true },
   264→   264→  })
   265→   265→  if (!booking) return NextResponse.json({ error: 'Booking not found' }, { status: 404 })
   266→   266→
   267→   267→  const billDays = parseInt(String(days)) || booking.days
   268→   268→  const actualRoomTotal = booking.ratePerDay * billDays
   269→   269→  const customRoom = parseFloat(String(billedRoomTotal))
   270→   270→  const billedRoom = !isNaN(customRoom) && customRoom > 0 ? customRoom : actualRoomTotal
   271→   271→
   272→   272→  let foodTotal = 0
   273→   273→  let pendingOrders: { id: string }[] = []
   274→   274→  if (includeFood !== false) {
   275→   275→    pendingOrders = await DB.foodOrder.findMany({
   276→   276→      where: { bookingId: booking.id, status: 'PENDING' },
   277→   277→      select: { id: true },
   278→   278→    })
   279→   279→    const agg = await DB.foodOrder.aggregate({
   280→   280→      where: { bookingId: booking.id, status: 'PENDING' },
   281→   281→      _sum: { total: true },
   282→   282→    })
   283→   283→    foodTotal = agg._sum.total || 0
   284→   284→  }
   285→   285→
   286→   286→  const extra = parseFloat(String(extraCharges)) || 0
   287→   287→  const disc = parseFloat(String(discount)) || 0
   288→   288→  const gstPct = parseFloat(String(gstPercent)) || 0
   289→   289→  const taxable = billedRoom + foodTotal + extra - disc
   290→   290→  const billedGst = Math.round(taxable * gstPct) / 100
   291→   291→  const grandTotal = Math.round((taxable + billedGst) * 100) / 100
   292→   292→
   293→   293→  const cash = parseFloat(String(payCash)) || 0
   294→   294→  const upi = parseFloat(String(payUpi)) || 0
   295→   295→  const card = parseFloat(String(payCard)) || 0
   296→   296→  const paidTotal = cash + upi + card
   297→   297→  if (Math.abs(paidTotal - grandTotal) > 0.01) {
   298→   298→    return NextResponse.json(
   299→   299→      { error: `Payment split (₹${paidTotal}) must equal bill total (₹${grandTotal})` },
   300→   300→      { status: 400 }
   301→   301→    )
   302→   302→  }
   303→   303→
   304→   304→  const today = new Date()
   305→   305→  const dateStr =
   306→   306→    today.getFullYear().toString() +
   307→   307→    String(today.getMonth() + 1).padStart(2, '0') +
   308→   308→    String(today.getDate()).padStart(2, '0')
   309→   309→  const countToday = await DB.bill.count({
   310→   310→    where: { createdAt: { gte: new Date(today.toISOString().slice(0, 10) + 'T00:00:00') } },
   311→   311→  })
   312→   312→  const billNumber = `INV-${dateStr}-${String(countToday + 1).padStart(3, '0')}`
   313→   313→
   314→   314→  const bill = await DB.bill.create({
   315→   315→    data: {
   316→   316→      billNumber,
   317→   317→      bookingId: booking.id,
   318→   318→      days: billDays,
   319→   319→      actualRoomTotal,
   320→   320→      billedRoomTotal: billedRoom,
   321→   321→      gstPercent: gstPct,
   322→   322→      actualGst: billedGst,
   323→   323→      foodTotal,
   324→   324→      extraCharges: extra,
   325→   325→      discount: disc,
   326→   326→      grandTotal,
   327→   327→      payCash: cash,
   328→   328→      payUpi: upi,
   329→   329→      payCard: card,
   330→   330→      isCorporate: billedRoom !== actualRoomTotal || !!booking.isCorporate,
   331→   331→      corporateName: corporateName ? String(corporateName) : booking.guest.company || null,
   332→   332→      gstNumber: gstNumber ? String(gstNumber) : booking.guest.gst || null,
   333→   333→      notes: notes ? String(notes) : null,
   334→   334→    },
   335→   335→  })
   336→   336→
   337→   337→  if (pendingOrders.length > 0) {
   338→   338→    await DB.foodOrder.updateMany({
   339→   339→      where: { id: { in: pendingOrders.map((o) => o.id) } },
   340→   340→      data: { status: 'ADDED_TO_BILL' },
   341→   341→    })
   342→   342→  }
   343→   343→
   344→   344→  // ===== ACCOUNT CREDIT BALANCING =====
   345→   345→  // Internal accounts credit ONLY actual room rate + billed GST (+ food).
   346→   346→  const ledgerEntries: {
   347→   347→    date: Date
   348→   348→    type: string
   349→   349→    category: string
   350→   350→    description: string
   351→   351→    amount: number
   352→   352→    method: string
   353→   353→    source: string
   354→   354→    refId: string
   355→   355→  }[] = []
   356→   356→
   357→   357→  if (actualRoomTotal > 0) {
   358→   358→    ledgerEntries.push({
   359→   359→      date: new Date(),
   360→   360→      type: 'INCOME',
   361→   361→      category: 'ROOM_RENT',
   362→   362→      description: `Room ${booking.room.number} rent (${billDays} day${billDays > 1 ? 's' : ''} @ ₹${booking.ratePerDay}) - ${booking.guest.name}`,
   363→   363→      amount: actualRoomTotal,
   364→   364→      method: 'SPLIT',
   365→   365→      source: 'AUTO',
   366→   366→      refId: bill.id,
   367→   367→    })
   368→   368→  }
   369→   369→  if (foodTotal > 0) {
   370→   370→    ledgerEntries.push({
   371→   371→      date: new Date(),
   372→   372→      type: 'INCOME',
   373→   373→      category: 'FOOD',
   374→   374→      description: `Food charges - Room ${booking.room.number} - ${booking.guest.name}`,
   375→   375→      amount: foodTotal,
   376→   376→      method: 'SPLIT',
   377→   377→      source: 'AUTO',
   378→   378→      refId: bill.id,
   379→   379→    })
   380→   380→  }
   381→   381→  if (billedGst > 0) {
   382→   382→    ledgerEntries.push({
   383→   383→      date: new Date(),
   384→   384→      type: 'INCOME',
   385→   385→      category: 'GST',
   386→   386→      description: `GST ${gstPct}% on bill ${billNumber}${billedRoom !== actualRoomTotal ? ' (on billed amount)' : ''}`,
   387→   387→      amount: billedGst,
   388→   388→      method: 'SPLIT',
   389→   389→      source: 'AUTO',
   390→   390→      refId: bill.id,
   391→   391→    })
   392→   392→  }
   393→   393→  if (ledgerEntries.length > 0) {
   394→   394→    await DB.ledgerEntry.createMany({ data: ledgerEntries })
   395→   395→  }
   396→   396→
   397→   397→  if (checkout !== false) {
   398→   398→    await DB.booking.update({
   399→   399→      where: { id: booking.id },
   400→   400→      data: { status: 'COMPLETED', actualCheckOut: new Date() },
   401→   401→    })
   402→   402→    await DB.room.update({ where: { id: booking.roomId }, data: { status: 'VACANT' } })
   403→   403→  }
   404→   404→
   405→   405→  return NextResponse.json({ ...bill, booking })
   406→   406→}

   612→   612→async function listLedger(req: NextRequest) {
   613→   613→  const { searchParams } = new URL(req.url)
   614→   614→  const date = searchParams.get('date')
   615→   615→  const from = searchParams.get('from')
   616→   616→  const to = searchParams.get('to')
   617→   617→  const type = searchParams.get('type')
   618→   618→
   619→   619→  let where: Record<string, unknown> = {}
   620→   620→  if (date) {
   621→   621→    const start = new Date(date + 'T00:00:00')
   622→   622→    const end = new Date(date + 'T23:59:59.999')
   623→   623→    where = { date: { gte: start, lte: end } }
   624→   624→  } else if (from || to) {
   625→   625→    where = {
   626→   626→      date: {
   627→   627→        ...(from && { gte: new Date(from + 'T00:00:00') }),
   628→   628→        ...(to && { lte: new Date(to + 'T23:59:59.999') }),
   629→   629→      },
   630→   630→    }
   631→   631→  }
   632→   632→  if (type) where = { ...where, type }
   633→   633→
   634→   634→  const entries = await DB.ledgerEntry.findMany({
   635→   635→    where,
   636→   636→    orderBy: { date: 'desc' },
   637→   637→    take: 500,
   638→   638→  })
   639→   639→  const totalIncome = entries.filter((e) => e.type === 'INCOME').reduce((s, e) => s + e.amount, 0)
   640→   640→  const totalExpense = entries.filter((e) => e.type === 'EXPENSE').reduce((s, e) => s + e.amount, 0)
   641→   641→  return NextResponse.json({
   642→   642→    entries,
   643→   643→    totalIncome,
   644→   644→    totalExpense,
   645→   645→    net: totalIncome - totalExpense,
   646→   646→  })
   647→   647→}

   669→   669→async function getStats() {
   670→   670→  const now = new Date()
   671→   671→  const startToday = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   672→   672→  const endToday = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59.999)
   673→   673→
   674→   674→  const [rooms, activeBookings, todayBills, todayLedger, pendingFood] = await Promise.all([
   675→   675→    DB.room.findMany({ select: { status: true, rate: true } }),
   676→   676→    DB.booking.findMany({ where: { status: 'ACTIVE' }, include: { guest: true, room: true } }),
   677→   677→    DB.bill.findMany({ where: { createdAt: { gte: startToday, lte: endToday } } }),
   678→   678→    DB.ledgerEntry.findMany({ where: { date: { gte: startToday, lte: endToday } } }),
   679→   679→    DB.foodOrder.aggregate({ where: { status: 'PENDING' }, _sum: { total: true } }),
   680→   680→  ])
   681→   681→
   682→   682→  const vacant = rooms.filter((r) => r.status === 'VACANT').length
   683→   683→  const occupied = rooms.filter((r) => r.status === 'OCCUPIED').length
   684→   684→  const maintenance = rooms.filter((r) => r.status === 'MAINTENANCE').length
   685→   685→  const todayRevenue = todayBills.reduce((s, b) => s + b.grandTotal, 0)
   686→   686→  const income = todayLedger.filter((e) => e.type === 'INCOME').reduce((s, e) => s + e.amount, 0)
   687→   687→  const expense = todayLedger.filter((e) => e.type === 'EXPENSE').reduce((s, e) => s + e.amount, 0)
   688→   688→  const potentialRevenue = rooms.filter((r) => r.status === 'OCCUPIED').reduce((s, r) => s + r.rate, 0)
   689→   689→
   690→   690→  return NextResponse.json({
   691→   691→    totalRooms: rooms.length,
   692→   692→    vacant,
   693→   693→    occupied,
   694→   694→    maintenance,
   695→   695→    occupancyPercent: rooms.length ? Math.round((occupied / rooms.length) * 100) : 0,
   696→   696→    activeGuests: activeBookings.length,
   697→   697→    activeBookings: activeBookings.map((b) => ({
   698→   698→      id: b.id,
   699→   699→      guestName: b.guest.name,
   700→   700→      roomNumber: b.room.number,
   701→   701→      checkIn: b.checkIn,
   702→   702→      days: b.days,
   703→   703→      ratePerDay: b.ratePerDay,
   704→   704→    })),
   705→   705→    todayRevenue,
   706→   706→    todayCash: todayBills.reduce((s, b) => s + b.payCash, 0),
   707→   707→    todayUpi: todayBills.reduce((s, b) => s + b.payUpi, 0),
   708→   708→    todayCard: todayBills.reduce((s, b) => s + b.payCard, 0),
   709→   709→    todayIncome: income,
   710→   710→    todayExpense: expense,
   711→   711→    todayNet: income - expense,
   712→   712→    pendingFoodAmount: pendingFood._sum.total || 0,
   713→   713→    potentialRevenue,
   714→   714→  })
   715→   715→}

   717→   717→// ============ ROUTER ============
   718→   718→async function route(
   719→   719→  req: NextRequest,
   720→   720→  method: 'GET' | 'POST' | 'PATCH' | 'DELETE'
   721→   721→): Promise<NextResponse> {
   722→   722→  const session: DbSession = await acquireDb()
   723→   723→  DB = session.client
   724→   724→  persistFn = session.persist
   725→   725→  const url = new URL(req.url)
   726→   726→  const segments = url.pathname.replace(/^\/api\/?/, '').split('/').filter(Boolean)
   727→   727→  const resource = segments[0] || ''
   728→   728→  let body: Record<string, unknown> = {}
   729→   729→  if (method === 'POST' || method === 'PATCH') {
   730→   730→    try {
   731→   731→      body = await req.json()
   732→   732→    } catch {
   733→   733→      body = {}
   734→   734→    }
   735→   735→  }
   736→   736→
   737→   737→  try {
   738→   738→    const result = await dispatch(req, method, url, body)
   739→   739→    if (method !== 'GET') {
   740→   740→      await persistFn()
   741→   741→    }
   742→   742→    return result
   743→   743→  } catch (e) {
   744→   744→    const message = e instanceof Error ? e.message : 'Internal server error'
   745→   745→    return NextResponse.json({ error: message }, { status: 500 })
   746→   746→  }
   747→   747→}
   748→   748→
   749→   749→async function dispatch(
   750→   750→  req: NextRequest,
   751→   751→  method: 'GET' | 'POST' | 'PATCH' | 'DELETE',
   752→   752→  url: URL,
   753→   753→  body: Record<string, unknown>
   754→   754→): Promise<NextResponse> {
   755→   755→  const segments = url.pathname.replace(/^\/api\/?/, '').split('/').filter(Boolean)
   756→   756→  const resource = segments[0] || ''
   757→   757→  switch (resource) {
   758→   758→      case 'rooms':
   759→   759→        if (method === 'GET') return await listRooms()
   760→   760→        if (method === 'PATCH') return await updateRoom(body)
   761→   761→        break
   762→   762→      case 'guests':
   763→   763→        if (method === 'GET' && url.searchParams.get('phone')) return await lookupGuest(req)
   764→   764→        if (method === 'GET') return await listGuests(req)
   765→   765→        if (method === 'POST') return await upsertGuest(body)
   766→   766→        break
   767→   767→      case 'bookings':
   768→   768→        if (method === 'GET') return await listBookings(req)
   769→   769→        if (method === 'POST') return await createBooking(body)
   770→   770→        if (method === 'PATCH') return await updateBooking(body)
   771→   771→        break
   772→   772→      case 'bills':
   773→   773→        if (method === 'GET') return await listBills(req)
   774→   774→        if (method === 'POST') return await createBill(body)
   775→   775→        break
   776→   776→      case 'menu':
   777→   777→        if (method === 'GET') return await listMenu()
   778→   778→        if (method === 'POST') return await createMenuItem(body)
   779→   779→        if (method === 'PATCH') return await updateMenuItem(body)
   780→   780→        if (method === 'DELETE') return await deleteMenuItem(req)
   781→   781→        break
   782→   782→      case 'orders':
   783→   783→        if (method === 'GET') return await listOrders(req)
   784→   784→        if (method === 'POST') return await createOrder(body)
   785→   785→        if (method === 'PATCH') return await updateOrder(body)
   786→   786→        break
   787→   787→      case 'staff':
   788→   788→        if (method === 'GET') return await listStaff()
   789→   789→        if (method === 'POST') return await createStaff(body)
   790→   790→        if (method === 'PATCH') return await updateStaff(body)
   791→   791→        break
   792→   792→      case 'staff-payments':
   793→   793→        if (method === 'GET') return await listStaffPayments(req)
   794→   794→        if (method === 'POST') return await createStaffPayment(body)
   795→   795→        break
   796→   796→      case 'ledger':
   797→   797→        if (method === 'GET') return await listLedger(req)
   798→   798→        if (method === 'POST') return await createLedgerEntry(body)
   799→   799→        break
   800→   800→      case 'stats':
   801→   801→        if (method === 'GET') return await getStats()
   802→   802→        break
   803→   803→  }
   804→   804→  return NextResponse.json({ error: `Not found: ${method} /api/${resource}` }, { status: 404 })
   805→   805→}
