   111→async function createBooking(body: Record<string, unknown>) {
   112→  const {
   113→    roomId, phone, name, company, gst, address, checkIn, checkOut,
   114→    guestCount, advance, isCorporate, notes,
   115→  } = body
   116→  if (!roomId || !phone || !name) {
   117→    return NextResponse.json({ error: 'Room, phone and name are required' }, { status: 400 })
   118→  }
   119→  const room = await DB.room.findUnique({ where: { id: String(roomId) } })
   120→  if (!room) return NextResponse.json({ error: 'Room not found' }, { status: 404 })
   121→  if (room.status === 'OCCUPIED') {
   122→    return NextResponse.json({ error: `Room ${room.number} is already occupied` }, { status: 400 })
   123→  }
   124→
   125→  const guest = await DB.guest.upsert({
   126→    where: { phone: String(phone) },
   127→    update: {
   128→      name: String(name),
   129→      ...(company !== undefined && { company: String(company) }),
   130→      ...(gst !== undefined && { gst: String(gst) }),
   131→      ...(address !== undefined && { address: String(address) }),
   132→    },
   133→    create: {
   134→      phone: String(phone),
   135→      name: String(name),
   136→      ...(company ? { company: String(company) } : {}),
   137→      ...(gst ? { gst: String(gst) } : {}),
   138→      ...(address ? { address: String(address) } : {}),
   139→    },
   140→  })
   141→
   142→  const checkInDate = checkIn ? new Date(String(checkIn)) : new Date()
   143→  const checkOutDate = checkOut ? new Date(String(checkOut)) : null
   144→  let days = 1
   145→  if (checkOutDate) {
   146→    const diff = Math.ceil((checkOutDate.getTime() - checkInDate.getTime()) / (1000 * 60 * 60 * 24))
   147→    days = Math.max(1, diff)
   148→  }
   149→
   150→  const booking = await DB.booking.create({
   151→    data: {
   152→      roomId: String(roomId),
   153→      guestId: guest.id,
   154→      checkIn: checkInDate,
   155→      checkOut: checkOutDate,
   156→      days,
   157→      guestCount: parseInt(String(guestCount)) || 1,
   158→      ratePerDay: room.rate,
   159→      advance: parseFloat(String(advance)) || 0,
   160→      isCorporate: !!isCorporate,
   161→      notes: notes ? String(notes) : null,
   162→    },
   163→    include: { room: true, guest: true },
   164→  })
   165→
   166→  await DB.room.update({ where: { id: String(roomId) }, data: { status: 'OCCUPIED' } })
   167→
   168→  const adv = parseFloat(String(advance)) || 0
   169→  if (adv > 0) {
   170→    await DB.ledgerEntry.create({
   171→      data: {
   172→        date: new Date(),
   173→        type: 'INCOME',
   174→        category: 'ADVANCE',
   175→        description: `Advance from ${name} (Room ${room.number})`,
   176→        amount: adv,
   177→        method: 'CASH',
   178→        source: 'AUTO',
   179→        refId: booking.id,
   180→      },
   181→    })
   182→  }
   183→  return NextResponse.json(booking)
   184→}

   186→async function updateBooking(body: Record<string, unknown>) {
   187→  const { id, action, checkOut } = body
   188→  const booking = await DB.booking.findUnique({ where: { id: String(id) }, include: { room: true } })
   189→  if (!booking) return NextResponse.json({ error: 'Booking not found' }, { status: 404 })
   190→
   191→  if (action === 'checkout') {
   192→    const pendingFood = await DB.foodOrder.aggregate({
   193→      where: { bookingId: String(id), status: 'PENDING' },
   194→      _sum: { total: true },
   195→    })
   196→    if ((pendingFood._sum.total || 0) > 0) {
   197→      return NextResponse.json(
   198→        { error: 'Pending food orders exist. Add them to bill or mark paid first.' },
   199→        { status: 400 }
   200→      )
   201→    }
   202→    const updated = await DB.booking.update({
   203→      where: { id: String(id) },
   204→      data: { status: 'COMPLETED', actualCheckOut: new Date() },
   205→    })
   206→    await DB.room.update({ where: { id: booking.roomId }, data: { status: 'VACANT' } })
   207→    return NextResponse.json(updated)
   208→  }
   209→
   210→  if (action === 'extend') {
   211→    const newCheckOut = checkOut ? new Date(String(checkOut)) : null
   212→    if (!newCheckOut) return NextResponse.json({ error: 'New checkout date required' }, { status: 400 })
   213→    const diff = Math.ceil(
   214→      (newCheckOut.getTime() - new Date(booking.checkIn).getTime()) / (1000 * 60 * 60 * 24)
   215→    )
   216→    const days = Math.max(1, diff)
   217→    const updated = await DB.booking.update({
   218→      where: { id: String(id) },
   219→      data: { checkOut: newCheckOut, days },
   220→    })
   221→    return NextResponse.json(updated)
   222→  }
   223→
   224→  if (action === 'cancel') {
   225→    const updated = await DB.booking.update({
   226→      where: { id: String(id) },
   227→      data: { status: 'CANCELLED', actualCheckOut: new Date() },
   228→    })
   229→    await DB.room.update({ where: { id: booking.roomId }, data: { status: 'VACANT' } })
   230→    return NextResponse.json(updated)
   231→  }
   232→
   233→  return NextResponse.json({ error: 'Unknown action' }, { status: 400 })
   234→}

   255→async function createBill(body: Record<string, unknown>) {
   256→  const {
   257→    bookingId, days, billedRoomTotal, gstPercent, extraCharges, discount,
   258→    payCash, payUpi, payCard, includeFood, corporateName, gstNumber, notes, checkout,
   259→  } = body
   260→
   261→  const booking = await DB.booking.findUnique({
   262→    where: { id: String(bookingId) },
   263→    include: { room: true, guest: true },
   264→  })
   265→  if (!booking) return NextResponse.json({ error: 'Booking not found' }, { status: 404 })
   266→
   267→  const billDays = parseInt(String(days)) || booking.days
   268→  const actualRoomTotal = booking.ratePerDay * billDays
   269→  const customRoom = parseFloat(String(billedRoomTotal))
   270→  const billedRoom = !isNaN(customRoom) && customRoom > 0 ? customRoom : actualRoomTotal
   271→
   272→  let foodTotal = 0
   273→  let pendingOrders: { id: string }[] = []
   274→  if (includeFood !== false) {
   275→    pendingOrders = await DB.foodOrder.findMany({
   276→      where: { bookingId: booking.id, status: 'PENDING' },
   277→      select: { id: true },
   278→    })
   279→    const agg = await DB.foodOrder.aggregate({
   280→      where: { bookingId: booking.id, status: 'PENDING' },
   281→      _sum: { total: true },
   282→    })
   283→    foodTotal = agg._sum.total || 0
   284→  }
   285→
   286→  const extra = parseFloat(String(extraCharges)) || 0
   287→  const disc = parseFloat(String(discount)) || 0
   288→  const gstPct = parseFloat(String(gstPercent)) || 0
   289→  const taxable = billedRoom + foodTotal + extra - disc
   290→  const billedGst = Math.round(taxable * gstPct) / 100
   291→  const grandTotal = Math.round((taxable + billedGst) * 100) / 100
   292→
   293→  const cash = parseFloat(String(payCash)) || 0
   294→  const upi = parseFloat(String(payUpi)) || 0
   295→  const card = parseFloat(String(payCard)) || 0
   296→  const paidTotal = cash + upi + card
   297→  if (Math.abs(paidTotal - grandTotal) > 0.01) {
   298→    return NextResponse.json(
   299→      { error: `Payment split (₹${paidTotal}) must equal bill total (₹${grandTotal})` },
   300→      { status: 400 }
   301→    )
   302→  }
   303→
   304→  const today = new Date()
   305→  const dateStr =
   306→    today.getFullYear().toString() +
   307→    String(today.getMonth() + 1).padStart(2, '0') +
   308→    String(today.getDate()).padStart(2, '0')
   309→  const countToday = await DB.bill.count({
   310→    where: { createdAt: { gte: new Date(today.toISOString().slice(0, 10) + 'T00:00:00') } },
   311→  })
   312→  const billNumber = `INV-${dateStr}-${String(countToday + 1).padStart(3, '0')}`
   313→
   314→  const bill = await DB.bill.create({
   315→    data: {
   316→      billNumber,
   317→      bookingId: booking.id,
   318→      days: billDays,
   319→      actualRoomTotal,
   320→      billedRoomTotal: billedRoom,
   321→      gstPercent: gstPct,
   322→      actualGst: billedGst,
   323→      foodTotal,
   324→      extraCharges: extra,
   325→      discount: disc,
   326→      grandTotal,
   327→      payCash: cash,
   328→      payUpi: upi,
   329→      payCard: card,
   330→      isCorporate: billedRoom !== actualRoomTotal || !!booking.isCorporate,
   331→      corporateName: corporateName ? String(corporateName) : booking.guest.company || null,
   332→      gstNumber: gstNumber ? String(gstNumber) : booking.guest.gst || null,
   333→      notes: notes ? String(notes) : null,
   334→    },
   335→  })
   336→
   337→  if (pendingOrders.length > 0) {
   338→    await DB.foodOrder.updateMany({
   339→      where: { id: { in: pendingOrders.map((o) => o.id) } },
   340→      data: { status: 'ADDED_TO_BILL' },
   341→    })
   342→  }
   343→
   344→  // ===== ACCOUNT CREDIT BALANCING =====
   345→  // Internal accounts credit ONLY actual room rate + billed GST (+ food).
   346→  const ledgerEntries: {
   347→    date: Date
   348→    type: string
   349→    category: string
   350→    description: string
   351→    amount: number
   352→    method: string
   353→    source: string
   354→    refId: string
   355→  }[] = []
   356→
   357→  if (actualRoomTotal > 0) {
   358→    ledgerEntries.push({
   359→      date: new Date(),
   360→      type: 'INCOME',
   361→      category: 'ROOM_RENT',
   362→      description: `Room ${booking.room.number} rent (${billDays} day${billDays > 1 ? 's' : ''} @ ₹${booking.ratePerDay}) - ${booking.guest.name}`,
   363→      amount: actualRoomTotal,
   364→      method: 'SPLIT',
   365→      source: 'AUTO',
   366→      refId: bill.id,
   367→    })
   368→  }
   369→  if (foodTotal > 0) {
   370→    ledgerEntries.push({
   371→      date: new Date(),
   372→      type: 'INCOME',
   373→      category: 'FOOD',
   374→      description: `Food charges - Room ${booking.room.number} - ${booking.guest.name}`,
   375→      amount: foodTotal,
   376→      method: 'SPLIT',
   377→      source: 'AUTO',
   378→      refId: bill.id,
   379→    })
   380→  }
   381→  if (billedGst > 0) {
   382→    ledgerEntries.push({
   383→      date: new Date(),
   384→      type: 'INCOME',
   385→      category: 'GST',
   386→      description: `GST ${gstPct}% on bill ${billNumber}${billedRoom !== actualRoomTotal ? ' (on billed amount)' : ''}`,
   387→      amount: billedGst,
   388→      method: 'SPLIT',
   389→      source: 'AUTO',
   390→      refId: bill.id,
   391→    })
   392→  }
   393→  if (ledgerEntries.length > 0) {
   394→    await DB.ledgerEntry.createMany({ data: ledgerEntries })
   395→  }
   396→
   397→  if (checkout !== false) {
   398→    await DB.booking.update({
   399→      where: { id: booking.id },
   400→      data: { status: 'COMPLETED', actualCheckOut: new Date() },
   401→    })
   402→    await DB.room.update({ where: { id: booking.roomId }, data: { status: 'VACANT' } })
   403→  }
   404→
   405→  return NextResponse.json({ ...bill, booking })
   406→}

   464→async function createOrder(body: Record<string, unknown>) {
   465→  const { bookingId, roomId, tableNo, items, notes } = body
   466→  if (!items || !Array.isArray(items) || items.length === 0) {
   467→    return NextResponse.json({ error: 'At least one item required' }, { status: 400 })
   468→  }
   469→  const total = items.reduce(
   470→    (s: number, it: { price: number; quantity: number }) => s + it.price * it.quantity,
   471→    0
   472→  )
   473→  const order = await DB.foodOrder.create({
   474→    data: {
   475→      bookingId: bookingId ? String(bookingId) : null,
   476→      roomId: roomId ? String(roomId) : null,
   477→      tableNo: tableNo ? String(tableNo) : null,
   478→      total,
   479→      notes: notes ? String(notes) : null,
   480→      items: {
   481→        create: items.map((it: { menuItemId?: string; name: string; price: number; quantity: number }) => ({
   482→          menuItemId: it.menuItemId || null,
   483→          name: String(it.name),
   484→          price: it.price,
   485→          quantity: it.quantity,
   486→        })),
   487→      },
   488→    },
   489→    include: { items: true, room: true, booking: { include: { guest: true } } },
   490→  })
   491→  return NextResponse.json(order)
   492→}

   549→      ...(name !== undefined && { name: String(name) }),
   550→      ...(phone !== undefined && { phone: String(phone) }),
   551→      ...(role !== undefined && { role: String(role) }),
   552→      ...(salary !== undefined && { salary: parseFloat(String(salary)) }),
   553→      ...(address !== undefined && { address: String(address) }),
   554→      ...(aadhaar !== undefined && { aadhaar: String(aadhaar) }),
   555→      ...(active !== undefined && { active: !!active }),
   556→    },
   557→  })

   574→async function createStaffPayment(body: Record<string, unknown>) {
   575→  const { staffId, type, amount, date, notes } = body
   576→  if (!staffId || !amount) {
   577→    return NextResponse.json({ error: 'Staff and amount required' }, { status: 400 })
   578→  }
   579→  const staff = await DB.staff.findUnique({ where: { id: String(staffId) } })
   580→  if (!staff) return NextResponse.json({ error: 'Staff not found' }, { status: 404 })
   581→
   582→  const amt = parseFloat(String(amount))
   583→  const payment = await DB.staffPayment.create({
   584→    data: {
   585→      staffId: String(staffId),
   586→      type: type ? String(type) : 'SALARY',
   587→      amount: amt,
   588→      date: date ? new Date(String(date)) : new Date(),
   589→      notes: notes ? String(notes) : null,
   590→    },
   591→    include: { staff: true },
   592→  })
   593→
   594→  if (payment.type === 'SALARY' || payment.type === 'BONUS') {
   595→    await DB.ledgerEntry.create({
   596→      data: {
   597→        date: payment.date,
   598→        type: 'EXPENSE',
   599→        category: 'SALARY',
   600→        description: `${payment.type === 'BONUS' ? 'Bonus' : 'Salary'} - ${staff.name}`,
   601→        amount: amt,
   602→        method: 'CASH',
   603→        source: 'AUTO',
   604→        refId: payment.id,
   605→      },
   606→    })
   607→  }
   608→  return NextResponse.json(payment)
   609→}

   611→// ============ LEDGER ============
   612→async function listLedger(req: NextRequest) {
   613→  const { searchParams } = new URL(req.url)
   614→  const date = searchParams.get('date')
   615→  const from = searchParams.get('from')
   616→  const to = searchParams.get('to')
   617→  const type = searchParams.get('type')
   618→
   619→  let where: Record<string, unknown> = {}
   620→  if (date) {
   621→    const start = new Date(date + 'T00:00:00')
   622→    const end = new Date(date + 'T23:59:59.999')
   623→    where = { date: { gte: start, lte: end } }
   624→  } else if (from || to) {
   625→    where = {
   626→      date: {
   627→        ...(from && { gte: new Date(from + 'T00:00:00') }),
   628→        ...(to && { lte: new Date(to + 'T23:59:59.999') }),
   629→      },
   630→    }
   631→  }
   632→  if (type) where = { ...where, type }
   633→
   634→  const entries = await DB.ledgerEntry.findMany({
   635→    where,
   636→    orderBy: { date: 'desc' },
   637→    take: 500,
   638→  })
   639→  const totalIncome = entries.filter((e) => e.type === 'INCOME').reduce((s, e) => s + e.amount, 0)
   640→  const totalExpense = entries.filter((e) => e.type === 'EXPENSE').reduce((s, e) => s + e.amount, 0)
   641→  return NextResponse.json({
   642→    entries,
   643→    totalIncome,
   644→    totalExpense,
   645→    net: totalIncome - totalExpense,
   646→  })
   647→}

   649→async function createLedgerEntry(body: Record<string, unknown>) {
   650→  const { type, category, description, amount, method, date } = body
   651→  if (!type || !amount || !description) {
   652→    return NextResponse.json({ error: 'Type, description and amount required' }, { status: 400 })
   653→  }
   654→  const entry = await DB.ledgerEntry.create({
   655→    data: {
   656→      type: type === 'INCOME' ? 'INCOME' : 'EXPENSE',
   657→      category: category ? String(category) : 'OTHER',
   658→      description: String(description),
   659→      amount: parseFloat(String(amount)),
   660→      method: method ? String(method) : 'CASH',
   661→      source: 'MANUAL',
   662→      date: date ? new Date(String(date)) : new Date(),
   663→    },
   664→  })
   665→  return NextResponse.json(entry)
   666→}

   668→// ============ STATS ============
   669→async function getStats() {
   670→  const now = new Date()
   671→  const startToday = new Date(now.getFullYear(), now.getMonth(), now.getDate())
   672→  const endToday = new Date(now.getFullYear(), now.getMonth(), now.getDate(), 23, 59, 59.999)
   673→
   674→  const [rooms, activeBookings, todayBills, todayLedger, pendingFood] = await Promise.all([
   675→    DB.room.findMany({ select: { status: true, rate: true } }),
   676→    DB.booking.findMany({ where: { status: 'ACTIVE' }, include: { guest: true, room: true } }),
   677→    DB.bill.findMany({ where: { createdAt: { gte: startToday, lte: endToday } } }),
   678→    DB.ledgerEntry.findMany({ where: { date: { gte: startToday, lte: endToday } } }),
   679→    DB.foodOrder.aggregate({ where: { status: 'PENDING' }, _sum: { total: true } }),
   680→  ])
   681→
   682→  const vacant = rooms.filter((r) => r.status === 'VACANT').length
   683→  const occupied = rooms.filter((r) => r.status === 'OCCUPIED').length
   684→  const maintenance = rooms.filter((r) => r.status === 'MAINTENANCE').length
   685→  const todayRevenue = todayBills.reduce((s, b) => s + b.grandTotal, 0)
   686→  const income = todayLedger.filter((e) => e.type === 'INCOME').reduce((s, e) => s + e.amount, 0)
   687→  const expense = todayLedger.filter((e) => e.type === 'EXPENSE').reduce((s, e) => s + e.amount, 0)
   688→  const potentialRevenue = rooms.filter((r) => r.status === 'OCCUPIED').reduce((s, r) => s + r.rate, 0)
   689→
   690→  return NextResponse.json({
   691→    totalRooms: rooms.length,
   692→    vacant,
   693→    occupied,
   694→    maintenance,
   695→    occupancyPercent: rooms.length ? Math.round((occupied / rooms.length) * 100) : 0,
   696→    activeGuests: activeBookings.length,
   697→    activeBookings: activeBookings.map((b) => ({
   698→      id: b.id,
   699→      guestName: b.guest.name,
   700→      roomNumber: b.room.number,
   701→      checkIn: b.checkIn,
   702→      days: b.days,
   703→      ratePerDay: b.ratePerDay,
   704→    })),
   705→    todayRevenue,
   706→    todayCash: todayBills.reduce((s, b) => s + b.payCash, 0),
   707→    todayUpi: todayBills.reduce((s, b) => s + b.payUpi, 0),
   708→    todayCard: todayBills.reduce((s, b) => s + b.payCard, 0),
   709→    todayIncome: income,
   710→    todayExpense: expense,
   711→    todayNet: income - expense,
   712→    pendingFoodAmount: pendingFood._sum.total || 0,
   713→    potentialRevenue,
   714→  })
   715→}

   717→// ============ ROUTER ============
   718→async function route(
   719→  req: NextRequest,
   720→  method: 'GET' | 'POST' | 'PATCH' | 'DELETE'
   721→): Promise<NextResponse> {
   722→  const session: DbSession = await acquireDb()
   723→  DB = session.client
   724→  persistFn = session.persist
   725→  const url = new URL(req.url)
   726→  const segments = url.pathname.replace(/^\/api\/?/, '').split('/').filter(Boolean)
   727→  const resource = segments[0] || ''
   728→  let body: Record<string, unknown> = {}
   729→  if (method === 'POST' || method === 'PATCH') {
   730→    try {
   731→      body = await req.json()
   732→    } catch {
   733→      body = {}
   734→    }
   735→  }
   736→
   737→  try {
   738→    const result = await dispatch(req, method, url, body)
   739→    if (method !== 'GET') {
   740→      await persistFn()
   741→    }
   742→    return result
   743→  } catch (e) {
   744→    const message = e instanceof Error ? e.message : 'Internal server error'
   745→    return NextResponse.json({ error: message }, { status: 500 })
   746→  }
   747→}
   748→
   749→async function dispatch(
   750→  req: NextRequest,
   751→  method: 'GET' | 'POST' | 'PATCH' | 'DELETE',
   752→  url: URL,
   753→  body: Record<string, unknown>
   754→): Promise<NextResponse> {
   755→  const segments = url.pathname.replace(/^\/api\/?/, '').split('/').filter(Boolean)
   756→  const resource = segments[0] || ''
   757→  switch (resource) {
   758→      case 'rooms':
   759→        if (method === 'GET') return await listRooms()
   760→        if (method === 'PATCH') return await updateRoom(body)
   761→        break
   762→      case 'guests':
   763→        if (method === 'GET' && url.searchParams.get('phone')) return await lookupGuest(req)
   764→        if (method === 'GET') return await listGuests(req)
   765→        if (method === 'POST') return await upsertGuest(body)
   766→        break
   767→      case 'bookings':
   768→        if (method === 'GET') return await listBookings(req)
   769→        if (method === 'POST') return await createBooking(body)
   770→        if (method === 'PATCH') return await updateBooking(body)
   771→        break
   772→      case 'bills':
   773→        if (method === 'GET') return await listBills(req)
   774→        if (method === 'POST') return await createBill(body)
   775→        break
   776→      case 'menu':
   777→        if (method === 'GET') return await listMenu()
   778→        if (method === 'POST') return await createMenuItem(body)
   779→        if (method === 'PATCH') return await updateMenuItem(body)
   780→        if (method === 'DELETE') return await deleteMenuItem(req)
   781→        break
   782→      case 'orders':
   783→        if (method === 'GET') return await listOrders(req)
   784→        if (method === 'POST') return await createOrder(body)
   785→        if (method === 'PATCH') return await updateOrder(body)
   786→        break
   787→      case 'staff':
   788→        if (method === 'GET') return await listStaff()
   789→        if (method === 'POST') return await createStaff(body)
   790→        if (method === 'PATCH') return await updateStaff(body)
   791→        break
   792→      case 'staff-payments':
   793→        if (method === 'GET') return await listStaffPayments(req)
   794→        if (method === 'POST') return await createStaffPayment(body)
   795→        break
   796→      case 'ledger':
   797→        if (method === 'GET') return await listLedger(req)
   798→        if (method === 'POST') return await createLedgerEntry(body)
   799→        break
   800→      case 'stats':
   801→        if (method === 'GET') return await getStats()
   802→        break
   803→  }
   804→  return NextResponse.json({ error: `Not found: ${method} /api/${resource}` }, { status: 404 })
   805→}

   807→export async function GET(req: NextRequest) {
   808→  return route(req, 'GET')

   810→export async function POST(req: NextRequest) {

   812→}
   813→export async function PATCH(req: NextRequest) {
   814→  return route(req, 'PATCH')
   815→}
   816→export async function DELETE(req: NextRequest) {
   817→  return route(req, 'DELETE')
